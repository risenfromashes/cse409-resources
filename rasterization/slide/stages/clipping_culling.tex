\section{Clipping \& Culling}

\subsection{Overview}
\begin{frame}{Clipping \& Culling Stage}
  \begin{columns}
    \begin{column}{0.55\textwidth}
      \begin{raybox}{Clipping \& Culling}
        \textbf{Input:} Assembled primitives in clip space \\
        \textbf{Output:} Visible primitives ready for rasterization

        \vspace{0.3cm}
        \textbf{Purpose:}
        \begin{itemize}
          \item Remove geometry outside the view frustum
          \item Eliminate back-facing triangles
          \item Reduce work for subsequent stages
          \item Prevent rendering artifacts
        \end{itemize}

        \textcolor{DarkGray}{\textbf{Configurable fixed-function stage}}

\subsection{View Frustum Clipping}
\begin{frame}{View Frustum \& Clipping Planes}
  \begin{columns}
    \begin{column}{0.45\textwidth}
      \begin{conceptbox}{View Frustum Planes}
        \textbf{Six clipping planes in clip space:}
        \begin{itemize}
          \item \textbf{Near:} $z \geq 0$ (or $z \geq -w$)
          \item \textbf{Far:} $z \leq w$
          \item \textbf{Left:} $x \geq -w$
          \item \textbf{Right:} $x \leq w$
          \item \textbf{Bottom:} $y \geq -w$
          \item \textbf{Top:} $y \leq w$
        \end{itemize}

        \vspace{0.2cm}
        After projection matrix transforms world coordinates to clip space $(x, y, z, w)$.
      \end{conceptbox}
    \end{column}

  \vspace{0.3cm}

  \begin{mathbox}{Clipping Test in Homogeneous Coordinates}
    For a vertex in clip space $(x, y, z, w)$, it's inside the frustum if:
    \begin{align}
      -w \leq x \leq w \quad \text{and} \quad -w \leq y \leq w \quad \text{and} \quad 0 \leq z \leq w
    \end{align}
    \textbf{Note:} DirectX uses $0 \leq z \leq w$, OpenGL uses $-w \leq z \leq w$
  \end{mathbox}
\end{frame}

\begin{frame}{Clipping Methods}
  \begin{columns}
    \begin{column}{0.5\textwidth}
      \begin{raybox}{Homogeneous Clipping}
        \textbf{Performed in 4D clip space}

        \vspace{0.2cm}
        \textbf{Advantages:}
        \begin{itemize}
          \item Linear plane equations: $ax + by + cz + dw = 0$
          \item Handles perspective correctly
          \item No division by zero issues
          \item Hardware implementation standard
        \end{itemize}

        \vspace{0.2cm}
        \textbf{Process:}
        \begin{enumerate}
          \item Test vertices against 6 planes
          \item Classify primitives by vertex positions
          \item Generate intersection points
          \item Create new triangulated polygons
        \end{enumerate}
      \end{raybox}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{conceptbox}{Screen Space Clipping}
        \textbf{After perspective divide (NDC)}

        \vspace{0.2cm}
        \textbf{Applications:}
        \begin{itemize}
          \item Viewport rectangle clipping
          \item Scissor test implementation
          \item Guard band optimization
          \item 2D UI element clipping
        \end{itemize}

        \vspace{0.2cm}
        \textbf{Characteristics:}
        \begin{itemize}
          \item Simple 2D algorithms (Cohen-Sutherland)
          \item Rectangular clip regions
          \item Per-pixel rejection possible
        \end{itemize}
      \end{conceptbox}
    \end{column}
  \end{columns}
\end{frame}

\subsection{Triangle Clipping}
\begin{frame}{Triangle Clipping Cases}
  \begin{columns}
    \begin{column}{0.4\textwidth}
      \begin{mathbox}{Classification \& Splitting}
        \textbf{Vertex classification:}
        \begin{itemize}
          \item \textbf{Inside:} All vertices inside frustum
          \item \textbf{Outside:} All vertices outside
          \item \textbf{Intersecting:} Mixed vertices
        \end{itemize}

        \vspace{0.2cm}
        \textbf{Edge-plane intersection:}
        \begin{align}
          t &= \frac{d_1}{d_1 - d_2} \\
          \mathbf{P_{int}} &= \mathbf{P_1} + t(\mathbf{P_2} - \mathbf{P_1})
        \end{align}

        Where $d_i = ax_i + by_i + cz_i + dw_i$
      \end{mathbox}
    \end{column}
    \begin{column}{0.6\textwidth}
      \begin{tikzpicture}[scale=1.0]

\begin{frame}{Sutherland-Hodgman Clipping Algorithm}
  \begin{columns}
    \begin{column}{0.5\textwidth}
      \begin{raybox}{Sequential Plane Clipping}
        \textbf{Process each plane in sequence}

        \vspace{0.2cm}
        \textbf{For each clipping plane:}
        \begin{enumerate}
          \item Start with current polygon
          \item Process each edge against plane
          \item Build output vertex list
          \item Use output as input for next plane
        \end{enumerate}

        \vspace{0.2cm}
        \textbf{Edge processing rules:}
        \begin{itemize}
          \item \textcolor{SecondaryColor}{Inside → Inside}: Output end vertex
          \item \textcolor{AccentColor}{Inside → Outside}: Output intersection
          \item \textcolor{ObjectColor}{Outside → Inside}: Output intersection + end
          \item \textcolor{DarkGray}{Outside → Outside}: Output nothing
        \end{itemize}
      \end{raybox}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{tikzpicture}[scale=0.8]
        % Step 1: Original triangle
        \node[font=\small, PrimaryColor] at (2.5,4.5) {\textbf{Step 1: Original Triangle}};

        (1,4) -- (4,4.2) -- (2.5,3.2) -- cycle;
        \node[below, font=\tiny] at (2.5,3) {\textbf{Input: 3 vertices}};

\subsection{Back-Face Culling}
\begin{frame}{Back-Face Culling}
  \begin{columns}
    \begin{column}{0.45\textwidth}
      \begin{conceptbox}{Back-Face Culling}
        \textbf{Purpose:} Remove triangles facing away from viewer

        \vspace{0.2cm}
        \textbf{Method:} Test triangle normal against view direction
        \begin{align}
          \text{Cull if: } \mathbf{n} \cdot \mathbf{v} < 0
        \end{align}

        \vspace{0.2cm}
        \textbf{Benefits:}
        \begin{itemize}
          \item 50\% triangle reduction (closed meshes)
          \item Reduced overdraw
          \item Less fragment shading
          \item Prevents z-fighting on coplanar faces
        \end{itemize}

        \vspace{0.2cm}
        \textbf{Winding order:} CCW = front face (OpenGL)
      \end{conceptbox}
    \end{column}
