\subsection{View Frustum Clipping}
\begin{frame}{View Frustum \& Clipping Planes}
  \footnotesize
  \begin{columns}
    \begin{column}{0.7\textwidth}
      \begin{conceptbox}{View Frustum Planes}
        \textbf{Six clipping planes in clip space:}
        \begin{itemize}
          \item \textbf{Near:}  $z \geq -w$
          \item \textbf{Far:} $z \leq w$
          \item \textbf{Left:} $x \geq -w$
          \item \textbf{Right:} $x \leq w$
          \item \textbf{Bottom:} $y \geq -w$
          \item \textbf{Top:} $y \leq w$
        \end{itemize}

        \vspace{0.2cm}
        After projection matrix transforms world coordinates to clip space $(x, y, z, w)$.
      \end{conceptbox}
    \end{column}

  \begin{mathbox}{Clipping Test in Homogeneous Coordinates}
    For a vertex in clip space $(x, y, z, w)$, it's inside the frustum if:
    \begin{align*}
      -w \leq x \leq w \quad \text{and} \quad -w \leq y \leq w \quad \text{and} \quad -w \leq z \leq w
    \end{align*}
  \end{mathbox}
\end{frame}

\begin{frame}{Clipping Methods: Homogeneous vs. Screen Space}
  \scriptsize
  \begin{columns}
    \begin{column}{0.5\textwidth}
      \begin{raybox}{Homogeneous Clipping}
        \textbf{Stage:} Before perspective divide (Clip Space)

        \vspace{0.2cm}
        \textbf{Process:}
        \begin{enumerate}
          \item Test vertices against six clip planes
          \item Classify primitives (inside, outside, partial)
          \item Compute intersection points if partially inside
          \item Triangulate resulting polygon
        \end{enumerate}

        \vspace{0.2cm}
        \textbf{Advantages:}
        \begin{itemize}
          \item Linear equations: $ax + by + cz + dw = 0$
          \item Standard in GPU hardware
        \end{itemize}

        \vspace{0.2cm}
        \textbf{Typical Use:} 3D scene rendering
      \end{raybox}
    \end{column}

    \begin{column}{0.5\textwidth}
      \begin{conceptbox}{Screen Space Clipping}
        \textbf{Stage:} After perspective divide (NDC)

        \vspace{0.2cm}
        \textbf{Process:}
        \begin{enumerate}
          \item Apply perspective divide
          \item Use 2D clipping algorithms (e.g., Cohen-Sutherland)
          \item Clip against screen-aligned rectangles
        \end{enumerate}

        \vspace{0.2cm}
        \textbf{Characteristics:}
        \begin{itemize}
          \item Simpler 2D math
          \item Often implemented per-pixel
          \item Good for scissor test, UI, and 2D overlays
        \end{itemize}

        \vspace{0.2cm}
        \textbf{Typical Use:} 2D GUI rendering
      \end{conceptbox}
    \end{column}
  \end{columns}

  \vspace{0.2cm}
  \centering
  \tiny{\textit{Note: Modern GPUs primarily perform homogeneous clipping in hardware, followed by optional screen-space techniques for specific tasks.}}
\end{frame}
\subsection{Triangle Clipping}
\begin{frame}{Triangle Clipping: Classification \& Intersection}
  \footnotesize
  \begin{columns}
    \begin{column}{0.55\textwidth}
      \only<1>{
        \begin{mathbox}{Triangle Classification}
          \scriptsize
          \textbf{Vertex classification:}
          \begin{itemize}
            \item \textbf{Fully Inside:} All vertices inside frustum $\Rightarrow$ Keep triangle
            \item \textbf{Fully Outside:} All vertices outside a single plane $\Rightarrow$ Discard
            \item \textbf{Intersecting:} Partially inside $\Rightarrow$ Clip and generate polygon
          \end{itemize}
        \end{mathbox}
      }
      \only<2>{
        \begin{mathbox}{Edge-Plane Intersection}
          For a clipping plane $ax + by + cz + dw = 0$, and an edge $(\mathbf{P}_1, \mathbf{P}_2)$:
          \begin{align*}
            d_i &= ax_i + by_i + cz_i + dw_i \\
            t &= \frac{d_1}{d_1 - d_2} \\
            \mathbf{P}_{\text{int}} &= \mathbf{P}_1 + t(\mathbf{P}_2 - \mathbf{P}_1)
          \end{align*}
          New intersection points lie on the clipping boundary.
        \end{mathbox}

        \begin{conceptbox}{Polygon Reconstruction}
          Clipping may produce convex polygons with 4+ vertices. \\
          These must be \textbf{triangulated} (e.g., using a fan from one vertex) before rasterization.
        \end{conceptbox}
      }
    \end{column}

    \begin{column}{0.45\textwidth}

\begin{frame}{Sutherland-Hodgman Clipping Algorithm}
  \footnotesize
  \begin{columns}
    \begin{column}{0.7\textwidth}
      \begin{raybox}{Sequential Plane Clipping}
        \textbf{Process each plane in sequence}

        \vspace{0.2cm}
        \textbf{For each clipping plane:}
        \begin{enumerate}
          \item Start with current polygon
          \item Process each edge against plane
          \item Build output vertex list
          \item Use output as input for next plane
        \end{enumerate}

        \vspace{0.2cm}
        \textbf{Edge processing rules:}
        \begin{itemize}
          \item \textcolor{SecondaryColor}{Inside → Inside}: Output end vertex
          \item \textcolor{AccentColor}{Inside → Outside}: Output intersection
          \item \textcolor{ObjectColor}{Outside → Inside}: Output intersection + end
          \item \textcolor{DarkGray}{Outside → Outside}: Output nothing
        \end{itemize}
      \end{raybox}
    \end{column}
    \begin{column}{0.3\textwidth}
      \begin{tikzpicture}[scale=0.8]

\subsection{Back-Face Culling}
\begin{frame}{Back-Face Culling}
  \footnotesize
  \begin{columns}
    \begin{column}{0.6\textwidth}
      \begin{conceptbox}{Back-Face Culling}
        \textbf{Purpose:} Remove triangles facing away from viewer

        \vspace{0.2cm}
        \textbf{Method:} Test triangle normal against view direction
        \begin{align*}
          \text{Cull if: } \mathbf{n} \cdot \mathbf{v} < 0
        \end{align*}

        \vspace{0.2cm}
        \textbf{Benefits:}
        \begin{itemize}
          \item 50\% triangle reduction (closed meshes)
          \item Reduced overdraw
          \item Less fragment shading
        \end{itemize}

      \end{conceptbox}
    \end{column}
