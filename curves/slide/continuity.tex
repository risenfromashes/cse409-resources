\section{Curve Continuity}

\begin{frame}{$C^0$ Continuity: Position only}
  \begin{columns}
    \begin{column}{0.5\textwidth}
      \begin{mathbox}{$C^0$ Continuity}
        \small
        The curves meet at the same point:
        \[
          \mathbf{P_1}(1) = \mathbf{P_2}(0)
        \]
        but tangents may differ.

        \vspace{0.3cm}

        \textbf{Properties:}
        \begin{itemize}
            \scriptsize
          \item No gaps in the curve
          \item Might have sharp change in direction (kink)
          \item Minimum requirement for connectedness
        \end{itemize}
      \end{mathbox}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{center}
        \begin{tikzpicture}[scale=1.0]
          \draw[->] (-0.5,0) -- (4,0) node[right] {$x$};
          \draw[->] (0,-0.5) -- (0,3) node[above] {$y$};

          % First curve segment
          \draw[thick, PrimaryColor] (0.5,0.5) .. controls (1.5,2.0) .. (2,1.5);

          % Second curve segment (different tangent)
          \draw[thick, SecondaryColor] (2,1.5) .. controls (2.2,2.5) .. (3.5,1.8);

          % Junction point
          \fill[ObjectColor] (2,1.5) circle (3pt);

          % Show kink with dashed lines
          \draw[dashed, AccentColor] (1.5,1.2) -- (2,1.5) -- (2.5,2.0);
          \node[right] at (2.6,2.0) {\scriptsize \textcolor{AccentColor}{Kink}};
        \end{tikzpicture}
      \end{center}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{$C^1$ Continuity: Tangent but curvature jump}
  \begin{columns}
    \begin{column}{0.5\textwidth}
      \begin{mathbox}{$C^1$ Continuity}
        \small
        Same position and same tangent:
        \begin{align*}
          \mathbf{P_1}(1) = \mathbf{P_2}(0), \\
          \mathbf{P_1}'(1) = \mathbf{P_2}'(0)
        \end{align*}
        but different curvature.

        \vspace{0.3cm}

        \textbf{Properties:}
        \begin{itemize}
            \scriptsize
          \item No visible kink
          \item Smooth tangent transition
        \end{itemize}
      \end{mathbox}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{center}
        \begin{tikzpicture}[scale=1.0]
          \draw[->] (-0.5,0) -- (4,0) node[right] {$x$};
          \draw[->] (0,-0.5) -- (0,3) node[above] {$y$};

          % First curve
          \draw[thick, PrimaryColor] (0.5,0.8) .. controls (1.5,2.0) .. (2,1.5);

          % Second curve - matched tangent but different curvature
          \draw[thick, SecondaryColor] (2,1.5) .. controls (2.5,1.0) .. (3.5,2.2);

          % Junction point
          \fill[ObjectColor] (2,1.5) circle (3pt);

          % Tangent arrows
          \draw[->, AccentColor, thick] (2,1.5) -- ++(0.35,-0.35);
          \draw[->, AccentColor, thick] (2,1.5) -- ++(-0.35,0.35);
          \node[above] at (2,1.8) {\scriptsize Same tangent};
        \end{tikzpicture}
      \end{center}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{$C^2$ Continuity: Tangent and curvature match}
  \begin{columns}
    \begin{column}{0.6\textwidth}
      \begin{mathbox}{$C^2$ Continuity}
        \small
        Same position, tangent, and curvature:
        \begin{align*}
          \mathbf{P_1}(1) = \mathbf{P_2}(0) \\
          \mathbf{P_1}'(1) = \mathbf{P_2}'(0) \\
          \mathbf{P_1}''(1) = \mathbf{P_2}''(0)
        \end{align*}

        \vspace{0.3cm}

        \textbf{Properties:}

        \begin{itemize}
            \scriptsize
          \item Perfectly smooth
          \item Highest visual quality
        \end{itemize}
      \end{mathbox}
    \end{column}
    \begin{column}{0.4\textwidth}
      \begin{center}
        \begin{tikzpicture}[scale=1.0]
          \draw[->] (-0.5,0) -- (3.5,0) node[right] {$x$};
          \draw[->] (0,-0.5) -- (0,3) node[above] {$y$};

          % Perfectly smooth curve
          \draw[thick, PrimaryColor]
          (0.5,1.5) .. controls (1.2,2.5) and (1.8,0.5) ..
          (3,1.5);

          % Junction (invisible in this smooth case)
          \fill[ObjectColor] (1.75,2.2) circle (2pt);

          % Curvature arc
          \draw[dashed, AccentColor] (1.75,2.2) circle (0.7);
          \node[above] at (1.75,3) {\scriptsize Same curvature};
        \end{tikzpicture}
      \end{center}
    \end{column}
  \end{columns}
\end{frame}
