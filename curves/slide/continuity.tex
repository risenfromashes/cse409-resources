\section{Curve Continuity}

\begin{frame}{What is Continuity?}
  \begin{columns}
    \begin{column}{0.5\textwidth}
      \begin{mathbox}{Mathematical Definition}
        \small
        Two curve segments $\mathbf{P_1}(t)$ and $\mathbf{P_2}(t)$ are continuous at their junction if they satisfy certain conditions.

        \vspace{0.3cm}
        \textbf{Levels of Continuity:}
        \begin{itemize}
            \scriptsize
          \item $C^0$: Position continuity
          \item $C^1$: Tangent continuity
          \item $C^2$: Curvature continuity
          \item $C^n$: $n$-th derivative continuity
        \end{itemize}
      \end{mathbox}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{center}
        \begin{tikzpicture}[scale=0.9]
          \draw[->] (-0.5,0) -- (4,0) node[right] {$t$};
          \draw[->] (0,-0.5) -- (0,3) node[above] {Position};

          % Discontinuous example
          \draw[thick, PrimaryColor] (0.5,0.5) -- (2,2);
          \draw[thick, SecondaryColor] (2,1) -- (3.5,2.5);

          % Highlight discontinuity
          \fill[AccentColor] (2,2) circle (2pt);
          \fill[AccentColor] (2,1) circle (2pt);
          \draw[AccentColor, thick, <->] (2,1) -- (2,2);

          \node[above] at (2,2.3) {\scriptsize \textcolor{AccentColor}{Gap}};
          \node[below] at (2,-0.7) {\scriptsize Not $C^0$ continuous};
        \end{tikzpicture}
      \end{center}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{$C^0$ Continuity: Position}
  \begin{columns}
    \begin{column}{0.5\textwidth}
      \begin{mathbox}{$C^0$ Continuity}
        \small
        The curves meet at the same point:
        \begin{align*}
          \mathbf{P_1}(1) = \mathbf{P_2}(0)
        \end{align*}

        \vspace{0.3cm}
        \textbf{Properties:}
        \begin{itemize}
            \scriptsize
          \item No gaps in the curve
          \item Connected but may have kinks
          \item Minimum requirement for smooth appearance
        \end{itemize}
      \end{mathbox}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{center}
        \begin{tikzpicture}[scale=1.0]
          \draw[->] (-0.5,0) -- (4,0) node[right] {$x$};
          \draw[->] (0,-0.5) -- (0,3) node[above] {$y$};

          % First curve segment
          \draw[thick, PrimaryColor] (0.5,0.5) .. controls (1.5,2.5) .. (2,1.5);

          % Second curve segment (C0 continuous)
          \draw[thick, SecondaryColor] (2,1.5) .. controls (2.5,0.5) .. (3.5,2);

          % Junction point
          \fill[ObjectColor] (2,1.5) circle (3pt);

          \node[above] at (2,1.8) {\scriptsize Junction};
          \node[below] at (2,-0.7) {\scriptsize $C^0$ Continuous};

          % Show the kink with angle markers
          \draw[dashed, AccentColor] (1.5,1.2) -- (2,1.5) -- (2.5,1.2);
          \node[right] at (2.7,1.2) {\scriptsize \textcolor{AccentColor}{Kink}};
        \end{tikzpicture}
      \end{center}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{$C^1$ Continuity: Tangent}
  \begin{columns}
    \begin{column}{0.5\textwidth}
      \begin{mathbox}{$C^1$ Continuity}
        \small
        Same position AND same tangent:
        \begin{align*}
          \mathbf{P_1}(1) &= \mathbf{P_2}(0) \\
          \mathbf{P_1}'(1) &= \mathbf{P_2}'(0)
        \end{align*}

        \vspace{0.3cm}
        \textbf{Properties:}
        \begin{itemize}
            \scriptsize
          \item No kinks or sharp corners
          \item Smooth tangent transition
          \item Visually pleasing curves
          \item Same velocity at junction
        \end{itemize}
      \end{mathbox}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{center}
        \begin{tikzpicture}[scale=1.0]
          \draw[->] (-0.5,0) -- (4,0) node[right] {$x$};
          \draw[->] (0,-0.5) -- (0,3) node[above] {$y$};

          % First curve segment
          \draw[thick, PrimaryColor] (0.5,0.8) .. controls (1.5,2.2) .. (2,1.5);

          % Second curve segment (C1 continuous)
          \draw[thick, SecondaryColor] (2,1.5) .. controls (2.5,0.8) .. (3.5,1.8);

          % Junction point
          \fill[ObjectColor] (2,1.5) circle (3pt);

          % Tangent vectors
          \draw[->, AccentColor, thick] (2,1.5) -- (2.3,1.2);
          \draw[->, AccentColor, thick] (2,1.5) -- (1.7,1.8);

          \node[above] at (2,1.8) {\scriptsize Same tangent};
          \node[below] at (2,-0.7) {\scriptsize $C^1$ Continuous};
        \end{tikzpicture}
      \end{center}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{$C^2$ Continuity: Curvature}
  \begin{columns}
    \begin{column}{0.6\textwidth}
      \begin{mathbox}{$C^2$ Continuity}
        \small
        Same position, tangent, AND curvature:
        \begin{align*}
          \mathbf{P_1}(1) &= \mathbf{P_2}(0) \\
          \mathbf{P_1}'(1) &= \mathbf{P_2}'(0) \\
          \mathbf{P_1}''(1) &= \mathbf{P_2}''(0)
        \end{align*}

        \vspace{0.3cm}
        \textbf{Properties:}
        \begin{itemize}
            \scriptsize
          \item No sudden change in bending
          \item Smoothest visual appearance
          \item Required for high-quality surfaces
          \item Same acceleration at junction
        \end{itemize}
      \end{mathbox}
    \end{column}
    \begin{column}{0.4\textwidth}
      \begin{center}
        \begin{tikzpicture}[scale=0.9]
          \draw[->] (-0.5,0) -- (3.5,0) node[right] {$x$};
          \draw[->] (0,-0.5) -- (0,3) node[above] {$y$};

          % Smooth curve (C2 continuous)
          \draw[thick, PrimaryColor] (0.5,1.5) .. controls (1.2,2.5) and (1.8,0.5) .. (3,1.5);

          % Junction point (invisible in this smooth case)
          \fill[ObjectColor] (1.75,1.5) circle (2pt);

          % Curvature visualization
          \draw[dashed, AccentColor] (1.75,1.5) circle (0.8);

          \node[above] at (1.75,2.5) {\scriptsize Same curvature};
          \node[below] at (1.75,-0.7) {\scriptsize $C^2$ Continuous};
        \end{tikzpicture}
      \end{center}
    \end{column}
  \end{columns}

  \vspace{0.5cm}

  \begin{conceptbox}{Why Continuity Matters}
    \footnotesize
    \textbf{Visual Quality:} Higher continuity = smoother, more natural appearance \\
    \textbf{Manufacturing:} Important for CNC machining, 3D printing \\
    \textbf{Animation:} Smooth motion paths require good continuity
  \end{conceptbox}
\end{frame}
