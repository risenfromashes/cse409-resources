\section{Bernstein Polynomials}

\begin{frame}{Bernstein Basis Functions}
  \begin{columns}
    \begin{column}{0.6\textwidth}
      \begin{mathbox}{Bernstein Polynomials}
        \small
        The $i$-th Bernstein polynomial of degree $n$:
        \begin{align*}
          B_{i,n}(t) = \binom{n}{i} t^i (1-t)^{n-i}
        \end{align*}

        \vspace{0.3cm}
        \textbf{Examples for degree 3:}
        \begin{align*}
          B_{0,3}(t) &= (1-t)^3 \\
          B_{1,3}(t) &= 3t(1-t)^2 \\
          B_{2,3}(t) &= 3t^2(1-t) \\
          B_{3,3}(t) &= t^3
        \end{align*}
      \end{mathbox}
    \end{column}
    \begin{column}{0.4\textwidth}
      \begin{center}
        \begin{tikzpicture}[scale=0.8]
          \draw[->] (-0.2,0) -- (4.2,0) node[right] {$t$};
          \draw[->] (0,-0.2) -- (0,3) node[above] {$B_{i,3}(t)$};

          % Plot Bernstein polynomials for degree 3
          \draw[thick, PrimaryColor] plot[domain=0:4, samples=50] (\x, {3*(1-\x/4)^3});
          \draw[thick, SecondaryColor] plot[domain=0:4, samples=50] (\x, {3*3*(\x/4)*(1-\x/4)^2});
          \draw[thick, AccentColor] plot[domain=0:4, samples=50] (\x, {3*3*(\x/4)^2*(1-\x/4)});
          \draw[thick, ObjectColor] plot[domain=0:4, samples=50] (\x, {3*(\x/4)^3});

          % Labels

\begin{frame}{Properties}
  \begin{columns}
    \begin{column}{0.6\textwidth}
      \begin{conceptbox}{Key Properties}
        \footnotesize
        \textbf{1. Non-negative:}
        \begin{align*}
          B_{i,n}(t)  \geq 0 \text{ for } t\in[0,1]
        \end{align*}
        {
          \scriptsize
          \emph{Justification:} For \(t\in[0,1]\) both \(t^i\) and \((1-t)^{n-i}\) are non‑negative. The binomial coefficient \(\binom{n}{i}\) is also non‑negative.
        }

        \vspace{0.6em}
        \textbf{2. Partition of unity:}
        \begin{align*}
          \sum_{i=0}^n B_{i,n}(t) = 1
        \end{align*}
        {
          \scriptsize
          \emph{Justification:}
          \begin{align*}
            \sum_{i=0}^n B_{i,n}(t) &=  \sum_{i=0}^n \binom{n}{i} t^i (1-t)^{n-i} \\
            &= (t + (1-t))^n = 1^n = 1
          \end{align*}
          The equality follows directly from the binomial theorem applied with \(x=t\) and \(y=1-t\).
        }
      \end{conceptbox}
    \end{column}

    \begin{column}{0.4\textwidth}
      \begin{center}
        \textbf{Non-negative}

% --- Slide 1: Convex Combination Basics ---
\begin{frame}{Bézier Curves as Convex Combinations}
  \begin{conceptbox}{Geometric Interpretation}
    \footnotesize
    \textbf{Convex Combination:} Since Bernstein polynomials are non-negative and sum to 1:
    \[
      \mathbf{B}(t) = \sum_{i=0}^{n} \underbrace{B_{i,n}(t)}_{\text{weights}} \mathbf{P}_i,
      \quad B_{i,n}(t) \ge 0,\quad \sum_{i=0}^n B_{i,n}(t) = 1
    \]
    Any point on the curve is a \emph{weighted average} of the control points.
  \end{conceptbox}

  \vspace{0.3cm}
  \begin{center}
    \begin{tikzpicture}[scale=0.9]
      % Control points
      \coordinate (P0) at (0.5,0.5);
      \coordinate (P1) at (1.5,3);
      \coordinate (P2) at (2.5,1);

      % Control polygon
      \draw[dashed, gray!50] (P0) -- (P1) -- (P2);

      % Bézier curve
      \draw[thick, PrimaryColor] (P0) .. controls (P1) .. (P2);

      % Control points with sizes according to weights
      \fill[SecondaryColor] (P0) circle (2pt);
      \fill[SecondaryColor] (P1) circle (4pt);
      \fill[SecondaryColor] (P2) circle (2pt);

      % Point on curve
      \coordinate (B) at (1.5,2.425);
      \fill[ObjectColor] (B) circle (2.5pt);

      % Labels
      \node[below] at (P0) {\scriptsize $w_0=0.25$};
      \node[above] at (P1) {\scriptsize $w_1=0.5$};
      \node[below] at (P2) {\scriptsize $w_2=0.25$};
      \node[right,ObjectColor,xshift=0.25cm] at (B) {\scriptsize Weighted avg.};
    \end{tikzpicture}
  \end{center}

  \vspace{0.2cm}
  \scriptsize
  The weights change smoothly with $t$, shifting the influence between control points.
\end{frame}

% --- Slide 2: Convex Hull Property ---
\begin{frame}{Convex Hull Property}
  \begin{conceptbox}{What is the Convex Hull?}
    \footnotesize
    The \textbf{convex hull} of a set of points is the smallest convex polygon containing them —
    imagine wrapping a rubber band around the points; the shape it makes is the convex hull.
  \end{conceptbox}
  \only<1>{
    \begin{figure}
      \centering
      \includegraphics[width=0.4\textwidth]{images/ConvexHull.svg.png}
      \caption{\textcolor{blue}{Convex hull} of a set of points}
    \end{figure}
  }
  \only<2>{

    \vspace{0.3cm}
    {
      \footnotesize
      \textbf{Why it applies here:}
      Bézier curve points are convex combinations of control points,
      \textbf{so they must lie inside the convex hull of the control polygon}.
    }
    \begin{center}
      \begin{tikzpicture}[scale=0.9]
        % Control points
        \coordinate (P0) at (0.5,0.5);
        \coordinate (P1) at (1.5,3);
        \coordinate (P2) at (2.5,1);
        \fill[fill=yellow!20, draw=none] (P0) -- (P1) -- (P2) -- cycle;

        % Control polygon
        \draw[dashed, gray!50] (P0) -- (P1) -- (P2);

        % Bézier curve
        \draw[thick, PrimaryColor] (P0) .. controls (P1) .. (P2);

        % Control points with sizes according to weights
        \fill[SecondaryColor] (P0) circle (2pt);
        \fill[SecondaryColor] (P1) circle (4pt);
        \fill[SecondaryColor] (P2) circle (2pt);

        % Point on curve
        \coordinate (B) at (1.5,2.425);
        \fill[ObjectColor] (B) circle (2.5pt);

        % Labels
        \node[below] at (P0) {\scriptsize $w_0=0.25$};
        \node[above] at (P1) {\scriptsize $w_1=0.5$};
        \node[below] at (P2) {\scriptsize $w_2=0.25$};
      \end{tikzpicture}
    \end{center}
    \scriptsize
    This is exactly the same idea as \textbf{barycentric coordinates} in a triangle:
    any point with non-negative barycentric coordinates lies inside the triangle.
  }
\end{frame}

% --- Slide 3: Consequences of Convex Hull Property ---
\begin{frame}{Why the Convex Hull Property Matters}
  \begin{columns}
    \begin{column}{0.6\textwidth}
      \begin{conceptbox}{Practical Consequences}
        \footnotesize
        \textbf{1. Natural Bounds:}
        \begin{itemize}
          \item Curve never leaves the convex hull
        \end{itemize}

        \textbf{2. Stability:}
        \begin{itemize}
          \item Small changes in control points $\rightarrow$

            small changes in curve
        \end{itemize}

        \textbf{3. Intuitive Control:}
        \begin{itemize}
          \item Smooth, predictable shaping
        \end{itemize}
      \end{conceptbox}
    \end{column}
