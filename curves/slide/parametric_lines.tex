\section{Parametric Lines}

\begin{frame}{Parametric Line Representation}
  \begin{columns}
    \begin{column}{0.6\textwidth}
      \begin{mathbox}{Linear Parametric Equation}
        Given two points $\mathbf{P_0}$ and $\mathbf{P_1}$:
        \begin{align*}
          \mathbf{P}(t) &= \mathbf{P_0} + t(\mathbf{P_1} - \mathbf{P_0}) \\
          &= (1-t)\mathbf{P_0} + t\mathbf{P_1}
        \end{align*}

        Where $t \in [0,1]$:
        \begin{itemize}
            \scriptsize
          \item $t = 0 \Rightarrow \mathbf{P}(0) = \mathbf{P_0}$
          \item $t = 1 \Rightarrow \mathbf{P}(1) = \mathbf{P_1}$
          \item $t = 0.5 \Rightarrow$ midpoint
        \end{itemize}
      \end{mathbox}
    \end{column}
    \begin{column}{0.4\textwidth}
      \begin{center}
        \begin{tikzpicture}[scale=1.0]
          % Coordinate system
          \draw[->] (-0.5,0) -- (4,0) node[right] {$x$};
          \draw[->] (0,-0.5) -- (0,3) node[above] {$y$};

          % Points
          \coordinate (P0) at (0.5,0.5);
          \coordinate (P1) at (3,2.5);

          % Line
          \draw[thick, PrimaryColor] (P0) -- (P1);

          % Mark points
          \fill[SecondaryColor] (P0) circle (3pt);
          \fill[SecondaryColor] (P1) circle (3pt);

          % Labels
          \node[below right] at (P0) {$\mathbf{P_0}$};
          \node[above right] at (P1) {$\mathbf{P_1}$};

          % Parameter positions
          \foreach \t/\pos in {0.25/{(1.125,1)}, 0.5/{(1.75,1.5)}, 0.75/{(2.375,2)}} {
            \fill[AccentColor] \pos circle (2pt);
          }

          % Parameter labels
          \node[right] at (1.125,0.8) {\scriptsize $t=0.25$};
          \node[right] at (1.75,1.3) {\scriptsize $t=0.5$};
          \node[right] at (2.375,1.8) {\scriptsize $t=0.75$};

          % Direction vector
          \draw[->, thick, ObjectColor] (P0) -- ++(0.4,0.32);
        \end{tikzpicture}
      \end{center}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{Linear Interpolation (LERP)}
  \begin{columns}
    \begin{column}{0.5\textwidth}
      \begin{mathbox}{LERP Function}
        \small
        Linear interpolation between two values:
        \begin{align*}
          \text{lerp}(A, B, t) &= (1-t)A + tB
        \end{align*}

        \textbf{Properties:}
        \begin{itemize}
            \scriptsize
          \item $\text{lerp}(A, B, 0) = A$
          \item $\text{lerp}(A, B, 1) = B$
          \item Linear relationship
        \end{itemize}
      \end{mathbox}

      \begin{conceptbox}{Code Example}
        \scriptsize
        \begin{minted}{python}
def lerp(A, B, t):
    return (1 - t) * A + t * B
        \end{minted}
      \end{conceptbox}
    \end{column}
