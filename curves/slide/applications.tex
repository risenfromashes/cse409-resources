\section{Applications}

\begin{frame}{Bézier Curves in Typography}
  \begin{columns}
    \begin{column}{0.5\textwidth}
      \begin{conceptbox}{Font Technologies}
        \footnotesize
        \textbf{TrueType Fonts:}
        \begin{itemize}
          \item Use quadratic Bézier curves
          \item Efficient rendering
          \item Compact file sizes
        \end{itemize}

        \textbf{OpenType/PostScript:}
        \begin{itemize}
          \item Use cubic Bézier curves
          \item Higher precision
          \item Better curve quality
        \end{itemize}

        \textbf{Advantages:}
        \begin{itemize}
          \item Scalable to any size
          \item Smooth curves at all resolutions
          \item Mathematical precision
        \end{itemize}
      \end{conceptbox}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{center}
        \textcolor{AccentColor}{[Font rendering example]}

        \vspace{0.5cm}

        \begin{tikzpicture}[scale=0.8]
          % Simulate letter 'S' construction

          .. controls (2,1.5) and (2,2.5) .. (1,2.5);

          % Control points
          \fill[SecondaryColor] (1,0.5) circle (1.5pt);

          \fill[AccentColor] (2,1.5) circle (1.5pt);

          \node[below] at (1,-0.2) {\scriptsize Letter 'S' with Bézier curves};
        \end{tikzpicture}
      \end{center}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Vector Graphics \& Design Tools}
  \begin{columns}
    \begin{column}{0.6\textwidth}
      \begin{conceptbox}{Software Applications}
        \scriptsize>
        \textbf{Adobe Illustrator:}
        \begin{itemize}
          \item Primary tool for vector graphics
          \item Pen tool creates Bézier paths
          \item Professional logo and illustration design
        \end{itemize}

        \textbf{SVG (Scalable Vector Graphics):}
        \begin{itemize}
          \item Web standard for vector graphics
          \item XML-based format using Bézier curves

          \item NURBS surfaces based on Bézier

\begin{frame}[fragile]{Animation \& User Interfaces}
  \begin{columns}
    \begin{column}{0.5\textwidth}
      \begin{conceptbox}{Motion Design}
        \footnotesize>
        \textbf{Easing Functions:}
        \begin{itemize}
          \item CSS cubic-bezier() function
          \item Smooth transitions and animations
          \item Natural motion feel
        \end{itemize}

        \textbf{Motion Paths:}
        \begin{itemize}
          \item Object trajectories in animation
          \item Camera movement in 3D
          \item Particle system paths
        \end{itemize}

        \textbf{UI Animations:}
        \begin{itemize}
          \item Button hover effects
          \item Page transitions
          \item Loading animations
        \end{itemize}
      \end{conceptbox}

      \begin{conceptbox}{CSS Example}
        \scriptsize>
        \begin{minted}{css}
.smooth-transition {
  transition: all 0.3s
    cubic-bezier(0.25, 0.1, 0.25, 1);
}

.custom-easing {
  animation: slide 1s
    cubic-bezier(0.68, -0.55, 0.265, 1.55);
}
        \end{minted}
      \end{conceptbox}
    \end{column}
    \begin{column}{0.5\textwidth}
