\section{Introduction}

\begin{frame}{Why Parametric Curves?}
  \begin{columns}
    \begin{column}{0.6\textwidth}
      \begin{conceptbox}{Problems with Traditional Representations}
        \scriptsize
        \textbf{Explicit form:} $y = f(x)$
        \begin{itemize}
          \item Single-valued functions only
          \item Cannot represent vertical lines
          \item Limited to function graphs
          \item No natural parameterization
        \end{itemize}

        \vspace{0.3cm}
        \textbf{Implicit form:} $F(x,y) = 0$
        \begin{itemize}
          \item Difficult to sample points
          \item Hard to control shape
          \item No direction information
          \item Complex for animations
        \end{itemize}
      \end{conceptbox}
    \end{column}
    \begin{column}{0.4\textwidth}
      \begin{center}
        \begin{tikzpicture}[scale=0.8]
          % Explicit function limitation
          \draw[->] (-0.5,0) -- (3,0) node[right] {$x$};
          \draw[->] (0,-0.5) -- (0,2.5) node[above] {$y$};

          \draw[thick, PrimaryColor] (0.2,0.3) .. controls (1,2) and (2,1) .. (2.8,2.2);
          \node[right] at (3,1.5) {\scriptsize $y = f(x)$};

          % Show vertical line problem
          \draw[thick, AccentColor, dashed] (1.5,0) -- (1.5,2.5);
          \node[above] at (1.5,2.7) {\scriptsize \textcolor{AccentColor}{Problem!}};

          \node[below] at (1.5,-0.8) {\scriptsize Cannot represent vertical lines};
        \end{tikzpicture}
      \end{center}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Parametric Representation: The Solution}
  \begin{center}
    \begin{tikzpicture}[scale=0.9]
      % Parametric curve example
      \draw[->] (-0.5,0) -- (4,0) node[right] {$x$};
      \draw[->] (0,-0.5) -- (0,3) node[above] {$y$};

      % Draw a parametric curve (circle)
      \draw[thick, PrimaryColor, ->] (2,1.5) circle (1);

      % Mark parameter positions
      \foreach \t/\pos in {0/{(3,1.5)}, 0.25/{(2,2.5)}, 0.5/{(1,1.5)}, 0.75/{(2,0.5)}} {
        \fill[SecondaryColor] \pos circle (2pt);
      }

      % Parameter labels
      \node[right] at (3.2,1.5) {\scriptsize $t=0$};
      \node[above] at (2,2.7) {\scriptsize $t=0.25$};
      \node[left] at (0.8,1.5) {\scriptsize $t=0.5$};
      \node[below] at (2,0.3) {\scriptsize $t=0.75$};

      \node[below] at (2,-1) {
        \begin{minipage}{6cm}
          \centering
          \begin{mathbox}{Parametric Form}
            \scriptsize
            $\mathbf{P}(t) =
            \begin{pmatrix} x(t) \\ y(t)
            \end{pmatrix}$
            Where $t \in [0,1]$ is the parameter
          \end{mathbox}
        \end{minipage}
      };
    \end{tikzpicture}
  \end{center}

  \begin{conceptbox}{Advantages of Parametric Curves}
    \footnotesize
    \begin{itemize}
      \item Can represent any curve shape (circles, loops, vertical lines)
      \item Natural parameterization for animations
      \item Easy to sample points uniformly
      \item Provides direction and velocity information
      \item Intuitive control through control points
    \end{itemize}
  \end{conceptbox}
\end{frame}
